3. Core Logic ImplementationA. The Safety Middleware (The Gatekeeper)This must be the very first function that touches user input before it reaches any LLM or emotion model.Python# app/core/safety.py
CRISIS_KEYWORDS = {"kill", "die", "suicide", "end it", "hurt myself"}

def run_safety_check(text_input: str) -> dict:
    """Detects immediate danger triggers."""
    if any(keyword in text_input.lower() for keyword in CRISIS_KEYWORDS):
        return {
            "status": "CRITICAL",
            "action": "TRIGGER_EMERGENCY_ESCALATION",
            "resources": "Tele-MANAS (14416), KIRAN (1800-599-0019)" #
        }
    return {"status": "SAFE", "action": "PROCEED"}
B. The Fusion EngineUse the weighted logic provided in your document to ensure the system is more robust than a single-modality model.Python# app/core/fusion.py
def calculate_fusion(face_score, voice_score, text_score):
    # Weights defined in Master Document: 0.45, 0.30, 0.25
    fused_score = (face_score * 0.45) + (voice_score * 0.30) + (text_score * 0.25)
    
    # Logic Override: If any modality is > 0.9 confidence in crisis, 
    # ignore weights and prioritize the highest risk score.
    if max(face_score, voice_score, text_score) > 0.9:
        return max(face_score, voice_score, text_score)
        
    return fused_score
